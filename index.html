/* ========================================== OPTIONS ========================================== */ if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders }); } /* ========================================== GET TEST ========================================== */ if (request.method === "GET") { return new Response( JSON.stringify({ success: true, service: "Seven7Studio API", status: "running" }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json; charset=UTF-8" } } ); } /* ========================================== ONLY POST ========================================== */ if (request.method !== "POST") { return new Response( JSON.stringify({ success: false, error: "Method Not Allowed" }), { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json; charset=UTF-8" } } ); } /* ========================================== MAIN ========================================== */ try { /* ---------------------------------------- CHECK SECRETS ---------------------------------------- */ if (!env.BOT_TOKEN) { throw new Error( "BOT_TOKEN تنظیم نشده است." ); } if (!env.ADMIN_CHAT_ID) { throw new Error( "ADMIN_CHAT_ID تنظیم نشده است." ); } /* ---------------------------------------- READ JSON ---------------------------------------- */ const data = await request.json(); /* ---------------------------------------- REQUEST TYPE ---------------------------------------- */ const type = String( data.type || "quote_request" ); /* ---------------------------------------- DATA ---------------------------------------- */ const packageName = String( data.package || "ثبت نشده" ).slice(0, 300); const name = String( data.name || "ثبت نشده" ).slice(0, 150); const phone = String( data.phone || "ثبت نشده" ).slice(0, 50); const date = String( data.date || "ثبت نشده" ).slice(0, 50); const time = String( data.time || "ثبت نشده" ).slice(0, 50); const location = String( data.location || "ثبت نشده" ).slice(0, 200); const ceremony = String( data.ceremony || "ثبت نشده" ).slice(0, 100); const description = String( data.description || "بدون توضیحات" ).slice(0, 1000); /* ---------------------------------------- TITLE ---------------------------------------- */ let title = "🔔 استعلام قیمت جدید"; if (type === "booking_request") { title = "📅 درخواست رزرو جدید"; } /* ---------------------------------------- TELEGRAM MESSAGE ---------------------------------------- */ const message = `${title} 👑 پکیج: ${packageName} 👤 نام: ${name} 📱 تماس: ${phone} 📅 تاریخ: ${date} ⏰ ساعت پیشنهادی: ${time} 📍 محل: ${location} 💍 نوع مراسم: ${ceremony} 📝 توضیحات: ${description}`; /* ---------------------------------------- TELEGRAM API ---------------------------------------- */ const telegramURL = `https://api.telegram.org/bot${env.BOT_TOKEN}/sendMessage`; const telegramResponse = await fetch( telegramURL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id: env.ADMIN_CHAT_ID, text: message, disable_web_page_preview: true }) } ); /* ---------------------------------------- TELEGRAM RESPONSE ---------------------------------------- */ let telegramResult; try { telegramResult = await telegramResponse.json(); } catch (error) { telegramResult = { ok: false, description: "پاسخ نامعتبر از Telegram API" }; charset=UTF-8" } } ); } /* ---------------------------------------- SUCCESS ---------------------------------------- */ return new Response( JSON.stringify({ success: true, message: "Request sent successfully" }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json; charset=UTF-8" } } ); } catch (error) { /* ======================================== SERVER ERROR ======================================== */ return new Response( JSON.stringify({ success: false, error: error.message || "Server error" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json; charset=UTF-8" } } ); } } };
